Редактируем файл (nano app.py). Новое содержание файла:
import logging
import os
from flask import Flask, request
from flasgger import Swagger

from controllers.welcome import welcome_bp
from controllers.calculator import calculator_bp


LOG_DIR = "/var/log/web-python-prj"
LOG_FILE = os.path.join(LOG_DIR, "app.log")

os.makedirs(LOG_DIR, exist_ok=True)

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
    handlers=[
        logging.FileHandler(LOG_FILE),
        logging.StreamHandler()
    ]
)

logger = logging.getLogger(__name__)


def create_app():
    app = Flask(__name__)

    app.register_blueprint(welcome_bp)
    app.register_blueprint(calculator_bp, url_prefix="/calculator")

    Swagger(
        app,
        template={
            "info": {
                "title": "Web Python API",
                "description": "Приветственная страница и калькулятор",
                "version": "1.0.0",
            }
        },
    )

    @app.before_request
    def log_request_info():
        logger.info("Request: %s %s", request.method, request.path)

    return app


app = create_app()

if __name__ == "__main__":
    logger.info("Application started")
    app.run(host="0.0.0.0", port=5000, debug=False)
